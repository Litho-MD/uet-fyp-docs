\section{Litho Template Manifest Specification}
Every publishable Litho template repository must place a declarative JSON manifest named \texttt{litho.json} at its root. The manifest defines author metadata, entrypoint \LaTeX\ files, required template variables, supported shortcodes, and license constraints.

\begin{verbatim}
{
  "$schema": 
    "https://litho-publishing.org/schemas/template-v1.json",
  "name": "uet-thesis-template",
  "version": "1.0.0",
  "displayName": 
    "UET Lahore CS Final Year Project Design Document Template",
  "description": 
    "Official thesis template conforming to UET guidelines.",
  "authors": [
    {
      "name": "Litho Core Team",
      "email": "templates@litho-publishing.org",
      "github": "Litho-MD"
    }
  ],
  "license": "MIT",
  "category": "Academic",
  "tags": ["uet", "thesis", "design-document", "academic"],
  "entrypoint": "uet_thesis.tex",
  "classFile": "uet_thesis.cls",
  "requiredVariables": [
    {
      "key": "title",
      "type": "string",
      "description": "Full document title"
    },
    {
      "key": "projectType",
      "type": "enum",
      "options": ["Development", "Research", "Hybrid", "Game"],
      "description": "Academic project classification"
    },
    {
      "key": "session",
      "type": "string",
      "example": "2023-2027",
      "description": "Academic session span"
    },
    {
      "key": "authors",

  "supportedShortcodes": [

      "name": "callout",

      "name": "gantt",

  ]
}
