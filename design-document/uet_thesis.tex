\newcommand{\diagram}[3][md]{%
  \begin{figure}[H]
    \centering
    \def\diagramsize{#1}%
    \ifx\diagramsize\diagramsizeDY
      % Use the full text block, reserving room for the caption and float spacing.
      \adjustbox{max width=\textwidth,max height=\dimexpr\textheight-4\baselineskip\relax}{\includegraphics{\detokenize{#2}}}%
    \else\ifx\diagramsize\diagramsizeSM
      \includegraphics[width=\textwidth,height=0.25\textheight,keepaspectratio]{\detokenize{#2}}
    \else\ifx\diagramsize\diagramsizeLG
      \includegraphics[width=\textwidth,height=0.7\textheight,keepaspectratio]{\detokenize{#2}}
    \else\ifx\diagramsize\diagramsizeFL
      \includegraphics[width=\textwidth,height=0.9\textheight,keepaspectratio]{\detokenize{#2}}
    \else
      \includegraphics[width=\textwidth,height=0.5\textheight,keepaspectratio]{\detokenize{#2}}
    \fi\fi\fi\fi
    \caption{#3}
    \label{fig:\detokenize{#2}}
  \end{figure}
}
