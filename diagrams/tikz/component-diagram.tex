% Component diagram for the Litho ecosystem.
% Included by Chapters/Chapter4.tex via \input{\detokenize{...}}.
% Interface IDs (INT-xx) refer to Table "Major interfaces" in Chapter 4.
%
% Helpers are defined here (outside the figure) so that macro parameters are
% not read as an argument of \resizebox. They only need TikZ + arrows.meta.

\tikzset{
  lc/.style={draw, line width=0.6pt, rectangle, align=center, font=\footnotesize,
             minimum height=1.3cm, inner sep=3pt},
  lcdep/.style={dashed, semithick, -{Triangle[open, length=2.4mm, width=2.2mm]}},
  lcflow/.style={line width=0.5pt, black!65, -{Stealth[length=1.8mm]}},
}

% Small UML component icon in the top-right corner of node #1.
\providecommand{\lcIcon}[1]{%
  \begin{scope}[shift={(#1.north east)}, xshift=-0.30cm, yshift=-0.11cm]
    \filldraw[fill=white, line width=0.4pt] (0,-0.26) rectangle (0.17,0);
    \filldraw[fill=white, line width=0.4pt] (-0.055,-0.06) rectangle (0.055,-0.10);
    \filldraw[fill=white, line width=0.4pt] (-0.055,-0.16) rectangle (0.055,-0.20);
  \end{scope}%
}

% Component: [style], name, position, text.
\providecommand{\lcComp}[4][]{%
  \node[lc, #1] (#2) at (#3) {#4};%
  \lcIcon{#2}%
}

% Secondary text line (technology / note) inside a component.
\providecommand{\lcSub}[1]{{\scriptsize\itshape\color{black!65}#1}}

% Small UML port: a square centred on a component edge at (#1,#2).
\providecommand{\lcPort}[2]{%
  \draw[fill=white, line width=0.4pt]
    (#1-0.11,#2-0.11) rectangle ++(0.22,0.22);%
}

% Ball-and-socket assembly connector on a vertical line at x = #1.
% #2 = y of the provider's edge (below), #3 = y of the consumer's edge (above),
% #4 = y of the ball centre.  The connector starts at the small port squares on
% both components, as in the tikz-uml "with port" assembly connector.
\providecommand{\lcIface}[4]{%
  \lcPort{#1}{#2}%
  \lcPort{#1}{#3}%
  \draw[semithick] (#1,#2+0.11) -- (#1,#4-0.16);
  \draw[semithick, fill=white] (#1,#4) circle (0.16);
  \draw[semithick] (#1-0.3,#4) arc[start angle=180, end angle=0, radius=0.3];
  \draw[semithick] (#1,#4+0.3) -- (#1,#3-0.11);%
}

% Interface label: anchor, x, y, id, text.
\providecommand{\lcLbl}[5]{%
  \node[font=\scriptsize, anchor=#1, align=left, inner sep=1pt] at (#2,#3)
       {\textbf{#4}\\#5};%
}
